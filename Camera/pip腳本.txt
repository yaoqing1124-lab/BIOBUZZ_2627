import cv2
import math
import numpy as np

# ==================== 可調參數 ====================
SCALE           = 0.5          # 處理縮放比例（0.5 = 半解析度，速度快 4 倍）
HSV_LOW         = np.array([20, 110, 110], dtype=np.uint8)   # 黃色下限
HSV_HIGH        = np.array([35, 255, 255], dtype=np.uint8)   # 黃色上限
MIN_AREA_FULL    = 200          # 最小面積（原始解析度像素），過濾雜訊
MAX_TARGETS      = 5            # 最多處理幾個目標
PIXELS_PER_UNIT  = 0.0          # 校正值：幾個像素 = 1 公分。0 = 只顯示像素長度
SHOW_MASK        = True         # 右下角顯示二值化縮圖（調 HSV 用）

# 長寬比限制設定（以 1:2 比例為中心，例如允許 1.5 到 2.5 之間，可自行調整容許範圍）
RATIO_MIN        = 1.5          
RATIO_MAX        = 2.5

# 這些只建立一次，避免每幀重複配置記憶體
KERNEL = cv2.getStructuringElement(cv2.MORPH_ELLIPSE, (3, 3))
FONT   = cv2.FONT_HERSHEY_SIMPLEX


def put_label(img, text, org, color, scale=0.45):
    """先畫黑邊再畫彩色字，任何背景都看得清楚"""
    cv2.putText(img, text, org, FONT, scale, (0, 0, 0), 3, cv2.LINE_AA)
    cv2.putText(img, text, org, FONT, scale, color, 1, cv2.LINE_AA)


# runPipeline() 每一幀都會被 Limelight 後端呼叫
def runPipeline(image, llrobot):
    # ---------- 1. 全部先初始化，保證一定有值 ----------
    largestContour = np.array([[]])
    llpython       = [0.0] * 8
    targets        = []          # 每個元素: (長, 寬, 角度, 中心x, 中心y, 面積, box點)
    ppu            = PIXELS_PER_UNIT

    # 影像保護
    if image is None or image.size == 0:
        return largestContour, image, llpython

    # 若後端給的是灰階，補成 3 通道，避免 cvtColor 崩潰
    if len(image.shape) == 2:
        image = cv2.cvtColor(image, cv2.COLOR_GRAY2BGR)

    h_full, w_full = image.shape[0], image.shape[1]

    # ---------- 2. 機器人可即時傳入校正值 llrobot[0] ----------
    if llrobot is not None and len(llrobot) > 0:
        try:
            v = float(llrobot[0])
            if v > 0.0:
                ppu = v
        except (TypeError, ValueError):
            ppu = PIXELS_PER_UNIT

    # ---------- 3. 縮小 → 模糊 → HSV → 二值化（效能關鍵） ----------
    small = cv2.resize(image, None, fx=SCALE, fy=SCALE,
                       interpolation=cv2.INTER_AREA)
    small = cv2.GaussianBlur(small, (3, 3), 0)
    hsv   = cv2.cvtColor(small, cv2.COLOR_BGR2HSV)
    mask  = cv2.inRange(hsv, HSV_LOW, HSV_HIGH)

    # 開運算殺小雜點，閉運算補破洞
    mask = cv2.morphologyEx(mask, cv2.MORPH_OPEN,  KERNEL, iterations=1)
    mask = cv2.morphologyEx(mask, cv2.MORPH_CLOSE, KERNEL, iterations=1)

    # ---------- 4. 找輪廓並依面積排序 ----------
    contours, _ = cv2.findContours(mask, cv2.RETR_EXTERNAL,
                                   cv2.CHAIN_APPROX_SIMPLE)

    min_area_small = MIN_AREA_FULL * SCALE * SCALE
    inv = 1.0 / SCALE

    scored = []
    for c in contours:
        a = cv2.contourArea(c)
        if a >= min_area_small:
            scored.append((a, c))          # tuple，排序時只比第一個元素
    scored.sort(key=lambda t: t[0], reverse=True)
    scored = scored[:MAX_TARGETS * 2]      # 多抓幾個來篩選比例，避免符合條件的不夠

    # ---------- 5. 對每個目標算最小外接旋轉矩形並篩選長寬比 ----------
    for area_small, c in scored:
        if len(targets) >= MAX_TARGETS:
            break

        (cx, cy), (rw, rh), _ang = cv2.minAreaRect(c)

        # 全部乘回原始解析度
        cx *= inv; cy *= inv; rw *= inv; rh *= inv

        length_px = max(rw, rh)             # 長邊 = 長度
        width_px  = min(rw, rh)             # 短邊 = 寬度
        
        # 計算長寬比 (長 / 寬)
        ratio = length_px / max(width_px, 1e-6)

        # 檢查是否符合 1:2 範圍 (這裡以 RATIO_MIN ~ RATIO_MAX 為例，代表 1:2 左右)
        if not (RATIO_MIN <= ratio <= RATIO_MAX):
            continue  # 不符合比例則跳過

        box = cv2.boxPoints(((cx, cy), (rw, rh), _ang)).astype(np.int32)

        # --- 找出「長軸」的兩個端點（兩條短邊的中點） ---
        e0 = math.hypot(box[1][0] - box[0][0], box[1][1] - box[0][1])
        e1 = math.hypot(box[2][0] - box[1][0], box[2][1] - box[1][1])
        if e0 >= e1:
            pa = ((box[1] + box[2]) * 0.5)
            pb = ((box[3] + box[0]) * 0.5)
        else:
            pa = ((box[0] + box[1]) * 0.5)
            pb = ((box[2] + box[3]) * 0.5)

        # 角度：影像 y 軸向下，所以取 -dy 讓逆時針為正，範圍 (-90, 90]
        dx = float(pb[0] - pa[0])
        dy = float(pb[1] - pa[1])
        angle_deg = math.degrees(math.atan2(-dy, dx))
        if angle_deg >  90.0: angle_deg -= 180.0
        if angle_deg <= -90.0: angle_deg += 180.0

        area_full = area_small * inv * inv
        targets.append((length_px, width_px, angle_deg, cx, cy,
                        area_full, box,
                        (int(pa[0]), int(pa[1])), (int(pb[0]), int(pb[1]))))

    # ---------- 6. 繪圖 ----------
    for i, t in enumerate(targets):
        length_px, width_px, angle_deg, cx, cy, area_full, box, pa, pb = t
        is_best = (i == 0)
        color = (0, 255, 0) if is_best else (0, 170, 255)
        thick = 2 if is_best else 1

        cv2.polylines(image, [box], True, color, thick, cv2.LINE_AA)

        if is_best:
            # 畫長軸（就是我們在量的那條線）
            cv2.line(image, pa, pb, (0, 0, 255), 2, cv2.LINE_AA)
            cv2.circle(image, pa, 4, (255, 0, 255), -1)
            cv2.circle(image, pb, 4, (255, 0, 255), -1)
            cv2.circle(image, (int(cx), int(cy)), 4, (255, 255, 255), -1)

            ratio = length_px / max(width_px, 1e-6)
            tx = int(cx) - 60
            ty = int(cy) - 14
            put_label(image, "L=%.1f px" % length_px, (tx, ty), (0, 255, 255), 0.55)
            put_label(image, "W=%.1f px  R=%.2f" % (width_px, ratio),
                      (tx, ty + 18), (0, 255, 255))
            put_label(image, "ang=%.1f deg" % angle_deg, (tx, ty + 34), (0, 255, 255))
            if ppu > 0.0:
                put_label(image, "%.2f cm" % (length_px / ppu),
                          (tx, ty + 52), (0, 255, 0), 0.6)
        else:
            put_label(image, "%.0f" % length_px,
                      (int(cx) - 15, int(cy)), (0, 170, 255), 0.4)

    # ---------- 7. HUD ----------
    put_label(image, "YELLOW LENGTH  targets=%d" % len(targets),
              (6, 18), (0, 255, 0), 0.5)
    if len(targets) == 0:
        put_label(image, "no target", (6, 36), (0, 0, 255), 0.5)

    # 右下角二值化縮圖，方便現場調 HSV
    if SHOW_MASK:
        tw, th = w_full // 4, h_full // 4
        if tw > 16 and th > 16:
            thumb = cv2.resize(mask, (tw, th), interpolation=cv2.INTER_NEAREST)
            image[h_full - th:h_full, w_full - tw:w_full] = \
                cv2.cvtColor(thumb, cv2.COLOR_GRAY2BGR)
            cv2.rectangle(image, (w_full - tw, h_full - th),
                          (w_full - 1, h_full - 1), (255, 255, 255), 1)

    # ---------- 8. 輸出 ----------
    if len(targets) > 0:
        length_px, width_px, angle_deg, cx, cy, area_full, _b, _pa, _pb = targets[0]
        real_len = (length_px / ppu) if ppu > 0.0 else 0.0
        llpython = [1.0, length_px, width_px, angle_deg,
                    cx, cy, real_len, float(len(targets))]
        # 輪廓乘回原始解析度後回傳
        # 這裡需要找到對應符合條件的原始輪廓，簡單起見我們用 scored 中符合的
        # 為了安全對應，我們直接抓 targets[0] 的 box 轉回或保留符合條件的 scored 索引
        # (此處配合原本邏輯微調)
        largestContour = scored[0][1].astype(np.float32) * inv
        largestContour = largestContour.astype(np.int32)

    return largestContour, image, llpython
